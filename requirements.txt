# Python 3.11+ (the engine uses sys.stdlib_module_names and modern f-strings).

                          # Chromium itself is NOT pip-installable: run
                          #   python3 -m playwright install chromium
PyMuPDF>=1.23             # measures the printed artifact (tools/check_scale.py Rule 4.3 gate)
pytest>=7.4               # tests/
